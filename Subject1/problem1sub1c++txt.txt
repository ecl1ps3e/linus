#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>
#include <sys/wait.h>

int main() {
    int p2c[2], c2p[2];

    if (pipe(p2c) == -1 || pipe(c2p) == -1) {
        perror("Pipe failed");
        return 1;
    }

    if (fork() == 0) {
        close(p2c[1]);
        close(c2p[0]);

        char filename[256];
        int n = read(p2c[0], filename, sizeof(filename) - 1);
        filename[n] = '\0';

        filename[strcspn(filename, "\n")] = '\0';
        close(p2c[0]);

        FILE *file = fopen(filename, "r");
        char response[256];

        if (file == NULL) {
            snprintf(response, sizeof(response), "Error: File '%s' does not exist.", filename);
        } else {
            int count = 0;
            char ch;
            while ((ch = fgetc(file)) != EOF) {
                if (ch == '\n') count++;
            }
            if (count > 0 || ftell(file) > 0) count++;

            fclose(file);
            snprintf(response, sizeof(response), "Lines in file: %d", count);
        }

        write(c2p[1], response, strlen(response));
        close(c2p[1]);
        exit(0);
    }

    close(p2c[0]);
    close(c2p[1]);

    char filename[256];
    printf("Enter filename: ");
    fgets(filename, sizeof(filename), stdin);

    write(p2c[1], filename, strlen(filename));
    close(p2c[1]);

    char response[256];
    int n = read(c2p[0], response, sizeof(response) - 1);
    response[n] = '\0';

    printf("Message from child: %s\n", response);
    close(c2p[0]);

    wait(NULL);
    return 0;
}
